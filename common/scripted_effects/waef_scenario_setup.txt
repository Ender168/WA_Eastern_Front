waef_clear_world_map = {
    every_state = {
        # Uniform map construction baseline.
        set_state_category = city
        set_building_level = { type = infrastructure level = 7 instant_build = yes }

        # No inherited industry outside the two player countries.
        set_building_level = { type = industrial_complex level = 0 instant_build = yes }
        set_building_level = { type = arms_factory level = 0 instant_build = yes }

        # Entire scenario is land-focused.
        set_building_level = { type = dockyard level = 0 instant_build = yes }
        set_building_level = {
            type = bunker
            level = 0
            province = { all_provinces = yes }
        }
        set_building_level = {
            type = coastal_bunker
            level = 0
            province = { all_provinces = yes }
        }
        set_building_level = { type = stronghold_network level = 0 instant_build = yes }

        # Remove every inherited supply hub before player hubs are rebuilt.
        set_building_level = {
            type = supply_node
            level = 0
            province = { all_provinces = yes }
        }
    }

    # WA state history gives these two Belgian states +2 slots.
    # Neutralize that so city = exactly 15 shared slots.
    6 = { add_extra_state_shared_building_slots = -2 }
    979 = { add_extra_state_shared_building_slots = -2 }
}

waef_normalize_player = {
    every_owned_state = {
        # Exactly one civilian and one military factory per playable state.
        set_building_level = { type = industrial_complex level = 1 instant_build = yes }
        set_building_level = { type = arms_factory level = 1 instant_build = yes }

        # Static state files are 550k; +175k gives 725k per state.
        add_manpower = 175000

        # One regional supply hub. The first VP in WA state history is the
        # regional capital/main city. If a state has no VP, fall back to one
        # random province so no playable state is left without a hub.
        add_building_construction = {
            type = supply_node
            level = 1
            instant_build = yes
            province = {
                all_provinces = yes
                limit_to_victory_point > 0
                first = yes
            }
        }
        if = {
            limit = {
                NOT = {
                    any_province_building_level = {
                        province = { all_provinces = yes }
                        building = supply_node
                        level > 0
                    }
                }
            }
            construct_building_in_random_province = { supply_node = 1 }
        }
    }
}
