waef_start_armament_fatigue = {
    set_country_flag = waef_expensive_armament_programme
    if = {
        limit = { NOT = { has_active_mission = waef_armament_fatigue } }
        activate_mission = waef_armament_fatigue
    }
}

# A repeated call safely keeps the same technologies; either completion order works.
waef_grant_german_jet_aircraft = {
    if = { limit = { has_completed_focus = WAEF_GER_C25 OR = { has_completed_focus = WAEF_GER_A3 } }
        if = { limit = { NOT = { has_dlc = "By Blood Alone" } } set_technology = { ger_jet_cv_fighter_1 = 1 ger_jet_fighter_multirole_1 = 1 } }
        if = { limit = { has_dlc = "By Blood Alone" } set_technology = { ger_jet_cv_fighter_ad_tech_1 = 1 ger_jet_fighter_multirole_ad_tech_1 = 1 ger_jet_fighter_multirole_ad_tech_1_2 = 1 } }
    }
    if = { limit = { has_completed_focus = WAEF_GER_C25 OR = { has_completed_focus = WAEF_GER_B3 } }
        if = { limit = { NOT = { has_dlc = "By Blood Alone" } } set_technology = { ger_jet_tactical_bomber_1 = 1 ger_jet_tactical_bomber_2 = 1 } }
        if = { limit = { has_dlc = "By Blood Alone" } set_technology = { ger_jet_fast_bomber_ad_tech_1 = 1 ger_jet_fast_bomber_ad_tech_2 = 1 } }
    }
    if = { limit = { has_completed_focus = WAEF_GER_C25 OR = { has_completed_focus = WAEF_GER_U3 } }
        if = { limit = { NOT = { has_dlc = "By Blood Alone" } } set_technology = { ger_jet_scout_plane_1 = 1 } }
        if = { limit = { has_dlc = "By Blood Alone" } set_technology = { ger_jet_fighter_ad_tech_1 = 1 ger_jet_scout_plane_ad_tech_1 = 1 } }
    }
}
