# Select by economic law first; an already-correct spirit must not fall through.
waef_sync_fuel_capacity_penalty = {
    if = {
        limit = {
            OR = {
                has_idea = civilian_economy
                has_idea = undisturbed_isolation
                has_idea = isolation
            }
        }
        remove_ideas = waef_reduced_fuel_capacity_early
        remove_ideas = waef_reduced_fuel_capacity_partial
        remove_ideas = waef_reduced_fuel_capacity_full
        if = {
            limit = { NOT = { has_idea = waef_reduced_fuel_capacity_civilian } }
            add_ideas = waef_reduced_fuel_capacity_civilian
        }
    }
    else_if = {
        limit = {
            has_idea = low_economic_mobilisation
        }
        remove_ideas = waef_reduced_fuel_capacity_civilian
        remove_ideas = waef_reduced_fuel_capacity_partial
        remove_ideas = waef_reduced_fuel_capacity_full
        if = {
            limit = { NOT = { has_idea = waef_reduced_fuel_capacity_early } }
            add_ideas = waef_reduced_fuel_capacity_early
        }
    }
    else_if = {
        limit = {
            has_idea = partial_economic_mobilisation
        }
        remove_ideas = waef_reduced_fuel_capacity_civilian
        remove_ideas = waef_reduced_fuel_capacity_early
        remove_ideas = waef_reduced_fuel_capacity_full
        if = {
            limit = { NOT = { has_idea = waef_reduced_fuel_capacity_partial } }
            add_ideas = waef_reduced_fuel_capacity_partial
        }
    }
    else_if = {
        limit = {
            OR = {
                has_idea = war_economy
                has_idea = tot_economic_mobilisation
                has_idea = over_mobilisation
                has_idea = exiled_economy
            }
        }
        remove_ideas = waef_reduced_fuel_capacity_civilian
        remove_ideas = waef_reduced_fuel_capacity_early
        remove_ideas = waef_reduced_fuel_capacity_partial
        if = {
            limit = { NOT = { has_idea = waef_reduced_fuel_capacity_full } }
            add_ideas = waef_reduced_fuel_capacity_full
        }
    }
    else = {
        # Unknown laws must never receive a potentially unsafe full penalty.
        remove_ideas = waef_reduced_fuel_capacity_civilian
        remove_ideas = waef_reduced_fuel_capacity_early
        remove_ideas = waef_reduced_fuel_capacity_partial
        remove_ideas = waef_reduced_fuel_capacity_full
    }
}
