on_actions = {
    on_daily_WEF = {
        effect = {
            waef_sync_fuel_capacity_penalty = yes
            waef_apply_scheduled_mobilization = yes

            if = {
                limit = {
                    date > 1941.1.29
                    NOT = { has_idea = waef_naval_doctrine_lock }
                }
                swap_ideas = {
                    remove_idea = waef_initial_doctrine_window
                    add_idea = waef_naval_doctrine_lock
                }
            }
        }
    }

    on_daily_EEF = {
        effect = {
            waef_sync_fuel_capacity_penalty = yes
            waef_apply_scheduled_mobilization = yes

            if = {
                limit = {
                    date > 1941.1.29
                    NOT = { has_idea = waef_naval_doctrine_lock }
                }
                swap_ideas = {
                    remove_idea = waef_initial_doctrine_window
                    add_idea = waef_naval_doctrine_lock
                }
            }
        }
    }
}
