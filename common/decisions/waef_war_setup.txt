waef_war_setup = {

    waef_war_preparation_countdown = {
        icon = generic_prepare_civil_war

        # Forced non-selectable mission, following World Ablaze patterns:
        # the country is allowed to own it, but it cannot start by itself
        # and cannot complete early through its goal condition.
        allowed = {
            OR = { tag = WEF tag = EEF }
        }

        activation = {
            always = no
        }

        available = {
            always = no
        }

        selectable_mission = no
        fire_only_once = yes
        is_good = no
        days_mission_timeout = 172

        timeout_effect = {
            remove_ideas = waef_prewar_truce
            set_country_flag = waef_war_unlocked
        }

        complete_effect = {
        }
    }

    waef_declare_scenario_war = {
        icon = generic_operation

        allowed = {
            OR = { tag = WEF tag = EEF }
        }

        visible = {
            has_global_flag = waef_scenario_1941
            has_country_flag = waef_war_unlocked
            NOT = { has_country_flag = waef_scenario_war_declared }
            has_war = no
        }

        available = {
            has_country_flag = waef_war_unlocked
            has_war = no
        }

        cost = 0
        fire_only_once = yes

        complete_effect = {
            add_timed_idea = {
                idea = waef_offensive_momentum
                days = 90
            }

            if = {
                limit = { tag = WEF }
                declare_war_on = {
                    target = EEF
                    type = annex_everything
                }
            }
            else_if = {
                limit = { tag = EEF }
                declare_war_on = {
                    target = WEF
                    type = annex_everything
                }
            }

            set_country_flag = waef_scenario_war_declared
        }

        ai_will_do = { base = 0 }
    }
}
